```java
@ApplicationScoped
public class UserInvitationService {
  
  @Inject UsuarioRepository usuarioRepo;
  @Inject UserInvitationRepository invitationRepo;
  @Inject MailService mailService;
  @Inject ConfigProvider config;
  
  public UserInvitationResponse inviteUser(CreateUserRequest req, Long invitedByUserId, String ipAddress) {
    // 1. Validate email
    if (!isValidEmail(req.getEmail())) {
      throw new InvalidEmailException("Email inválido");
    }
    
    // 2. Generate username: {3name}{3lastname}{3role}@control.ttcs
    String username = generateUsername(req.getNombre(), req.getApellido(), req.getRol());
    
    // 3. Check username uniqueness
    if (usuarioRepo.existsByUsername(username)) {
      throw new UsernameAlreadyExistsException("Usuario " + username + " ya existe");
    }
    
    // 4. Generate temporary password (12 chars)
    String tempPassword = generateTemporaryPassword();
    
    // 5. Hash password
    String hashedPassword = passwordService.hash(tempPassword);
    
    // 6. Create usuario record
    Usuario usuario = new Usuario();
    usuario.setUsername(username);
    usuario.setPassword(hashedPassword);
    usuario.setEmail(req.getEmail()); // personal email
    usuario.setNombre(req.getNombre());
    usuario.setApellido(req.getApellido());
    usuario.setRol(req.getRol());
    usuario.setStatus("ACTIVE");
    usuario = usuarioRepo.persist(usuario);
    
    // 7. Create invitation record
    UserInvitation invitation = new UserInvitation();
    invitation.setUsuarioId(usuario.getId());
    invitation.setEmailPersonal(req.getEmail());
    invitation.setInvitedAt(Instant.now());
    invitation.setInvitedBy(invitedByUserId);
    invitation.setStatus("INVITED");
    invitation.setIpAddress(ipAddress);
    invitation.setExpiryAt(Instant.now().plus(Duration.ofDays(30)));
    invitationRepo.persist(invitation);
    
    // 8. Send SMTP email (HTML template)
    mailService.sendInvitationEmail(
      to = req.getEmail(),
      username = username,
      temporaryPassword = tempPassword,
      loginUrl = config.getValue("app.login-url", String.class)
    );
    
    // 9. Audit log
    auditService.log("USER_INVITED", usuario.getId(), invitedByUserId, ipAddress);
    
    // 10. Return response (never expose temp password in response body after send)
    return new UserInvitationResponse(
      usuarioId = usuario.getId(),
      username = username,
      emailPersonal = req.getEmail(),
      status = "INVITED"
    );
  }
  
  private String generateUsername(String nombre, String apellido, String rol) {
    String n = nombre.substring(0, Math.min(3, nombre.length())).toLowerCase();
    String a = apellido.substring(0, Math.min(3, apellido.length())).toLowerCase();
    String r = rol.substring(0, Math.min(3, rol.length())).toLowerCase();
    return n + a + "." + r + "@control.ttcs";
  }
  
  private String generateTemporaryPassword() {
    SecureRandom random = new SecureRandom();
    String chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789!@#$%^&*";
    StringBuilder sb = new StringBuilder();
    for (int i = 0; i < 12; i++) {
      sb.append(chars.charAt(random.nextInt(chars.length())));
    }
    return sb.toString();
  }
  
  private boolean isValidEmail(String email) {
    String regex = "^[A-Za-z0-9+_.-]+@(.+)$";
    return email.matches(regex);
  }
}
```

**Acceptance Criteria:**
- [x] Username generated: {3name}{3lastname}{3role}@control.ttcs — formato real sin el punto (port exacto del backend Spring); ver D de apply-progress
- [ ] Temp password: 12 chars with entropy — **NO implementada, declarada en apply-progress.md (D.2)**: la cuenta nace PENDIENTE_INVITACION y el login la bloquea, asi que una temporal no serviria para entrar; el link de activacion es la credencial. Hace falta una pantalla de cambio de contrasena (o `must_change_password`), que es otro change de producto.
- [x] Usuario created in DB
- [x] Invitation record created
- [x] Email sent (verify in MailService)
- [x] Audit log recorded
- [x] No temp password in response body

### T-AUTH-INV-02.3: Create UserInvitationResource endpoint

**File:** `backend/electoral-api-quarkus/src/main/java/ec/gob/cne/electoralsystem/api/user/UserInvitationResource.java`

**Code:**
```java
@Path("api/v1/users")
@Produces(MediaType.APPLICATION_JSON)
@RequiredRole("MASTER")
public class UserInvitationResource {
  
  @Inject UserInvitationService invitationService;
  
  @POST
  @Path("invite")
  @RateLimit(requests = 1, windowSec = 5, identifier = "ip")
  public Response inviteUser(
    CreateUserRequest req,
    @Context SecurityContext sec,
    @Context HttpServletRequest http
  ) {
    Long invitedByUserId = (Long) sec.getUserPrincipal();
    String ipAddress = extractClientIp(http);
    
    UserInvitationResponse response = invitationService.inviteUser(req, invitedByUserId, ipAddress);
    return Response.ok(response).build();
  }
  
  private String extractClientIp(HttpServletRequest req) {
    String xfwd = req.getHeader("X-Forwarded-For");
    return xfwd != null ? xfwd.split(",")[0].trim() : req.getRemoteAddr();
  }
}
```

**Acceptance Criteria:**
- [x] POST /api/v1/users/invite requires MASTER role
- [x] Rate limit: 1/5s per IP
- [x] Response: { usuarioId, username, emailPersonal, status }
- [x] Status code: 200 OK on success
- [x] Status code: 429 on rate limit
- [x] Status code: 401 if not MASTER — 403 para un autenticado sin permiso (el contrato del repo) y 401 para quien no tiene sesion

```typescript
          <mat-option value="COORDINATOR">Coordinador</mat-option>
          <mat-option value="ADMIN">Administrador</mat-option>
        </mat-select>
      </mat-form-field>

      <button mat-raised-button color="primary" type="submit" [disabled]="form.invalid || loading">
        Enviar Invitación
      </button>
    </form>

    <div *ngIf="successMessage" class="success-alert">
      ✓ Invitación enviada a {{ successMessage }}
    </div>
  `

export class CreateUserFormComponent {
  form: FormGroup;
  loading = false;
  successMessage = '';

  constructor(
    private fb: FormBuilder,
    private userService: UserService,
    private router: Router
  ) {
    this.form = this.fb.group({
      nombre: ['', Validators.required],
      apellido: ['', Validators.required],
      email: ['', [Validators.required, Validators.email]],
      rol: ['', Validators.required]
    });
  }

  onSubmit() {
    if (this.form.invalid) return;

    this.loading = true;
    this.userService.inviteUser(this.form.value).subscribe({
      next: (response) => {
        this.successMessage = this.form.get('email')!.value;
        this.form.reset();
        this.loading = false;
      },
      error: (err) => {
        console.error('Error invitando usuario:', err);
        this.loading = false;
      }
    });
  }
}
```
