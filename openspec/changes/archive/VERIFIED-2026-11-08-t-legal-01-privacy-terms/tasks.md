  id BIGINT PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,

  email VARCHAR(255) NOT NULL,
  subject VARCHAR(255) NOT NULL,
  message TEXT NOT NULL,
  submitted_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  ip_address VARCHAR(45),
  status VARCHAR(50) DEFAULT 'NEW',

  responded_at TIMESTAMP
);

CREATE INDEX idx_dpo_status ON dpo_contact_requests(status);
CREATE INDEX idx_dpo_submitted ON dpo_contact_requests(submitted_at);

-- Create user export log table (GDPR Art. 15)
CREATE TABLE user_export_log (

  usuario_id BIGINT NOT NULL,
  export_id VARCHAR(36) UNIQUE NOT NULL,
  exported_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  ip_address VARCHAR(45),
  FOREIGN KEY (usuario_id) REFERENCES usuario(id) ON DELETE CASCADE
);

CREATE INDEX idx_export_user ON user_export_log(usuario_id);
CREATE INDEX idx_export_id ON user_export_log(export_id);

### T-LEGAL-01.3: Create UserLegalService for acceptance + export

**Description:** Implement business logic for accepting terms, exporting data, recording audits.

**File:** `backend/electoral-api-quarkus/src/main/java/ec/gob/cne/electoralsystem/api/user/UserLegalService.java`

**Methods:**
```java
@ApplicationScoped
public class UserLegalService {
  
  @Inject TermsAcceptanceAuditRepository auditRepo;
  @Inject UsuarioRepository usuarioRepo;
  
  // Accept new terms version
  public void acceptTerms(Long userId, String version, HttpServletRequest req) {
    Usuario user = usuarioRepo.findById(userId);
    String ip = extractClientIp(req);
    String userAgent = req.getHeader("User-Agent");
    
    // Insert audit record
    TermsAcceptanceAudit audit = new TermsAcceptanceAudit();
    audit.setUsuarioId(userId);
    audit.setTermsVersion(version);
    audit.setAcceptedAt(Instant.now());
    audit.setIpAddress(ip);
    audit.setUserAgent(userAgent);
    audit.setLocale(req.getLocale().toString());
    auditRepo.persist(audit);
    
    // Update usuario
    user.setAcceptedTermsVersion(version);
    user.setAcceptedTermsAt(Instant.now());
    usuarioRepo.update(user);
  }
  
  // Export user data (GDPR Art. 15)
  public Map<String, Object> exportUserData(Long userId) {
    Usuario user = usuarioRepo.findById(userId);
    List<Acta> actas = actaRepository.findByUsuario(userId);
    List<Dispute> disputes = disputeRepository.findByUsuario(userId);
    List<UserExportLog> exports = exportLogRepository.findByUsuario(userId);
    
    return Map.of(
      "exported_at", Instant.now(),
      "export_request_id", UUID.randomUUID().toString(),
      "user", userToMap(user),
      "actas_registradas", actas,
      "dispute_submissions", disputes,
      "exports_data", exports
    );
  }
  
  // Record export request (rate limiting handled by filter)
  public void logExport(Long userId, String exportId, HttpServletRequest req) {
    UserExportLog log = new UserExportLog();
    log.setUsuarioId(userId);
    log.setExportId(exportId);
    log.setIpAddress(extractClientIp(req));
    exportLogRepository.persist(log);
  }
  
  private String extractClientIp(HttpServletRequest req) {
    String xfwd = req.getHeader("X-Forwarded-For");
    return xfwd != null ? xfwd.split(",")[0].trim() : req.getRemoteAddr();
  }
}
```

**Acceptance Criteria:**
- [ ] acceptTerms() inserts audit record with version, timestamp, IP, user-agent
- [ ] acceptTerms() updates usuario.accepted_terms_version and accepted_terms_at
- [ ] exportUserData() returns complete JSON with user + actas + disputes
- [ ] exportUserData() includes export_request_id (UUID)
- [ ] logExport() records export in user_export_log table
- [ ] IP extraction handles X-Forwarded-For header (for proxies)

**Mutation:** Created UserLegalService with acceptance + export logic.

### T-LEGAL-01.4: Create endpoints for terms acceptance + export + DPO

**Description:** Implement POST `/api/v1/user/accept-terms`, GET `/api/v1/user/export-data`, POST `/api/v1/legal/contact-dpo`

**File:** `backend/electoral-api-quarkus/src/main/java/ec/gob/cne/electoralsystem/api/legal/UserLegalResource.java`

**Code:**
```java
@Path("api/v1/user")
@Produces(MediaType.APPLICATION_JSON)
public class UserLegalResource {
  
  @Inject UserLegalService legalService;
  @Inject UsuarioRepository usuarioRepo;
  @Inject ConfigProvider config;
  
  @POST
  @Path("accept-terms")
  @Authenticated
  public Response acceptTerms(@Context SecurityContext sec, @Context HttpServletRequest req) {
    Long userId = (Long) sec.getUserPrincipal();
    String version = config.getValue("legal.terms-version", String.class);
    
    legalService.acceptTerms(userId, version, req);
    return Response.ok(Map.of("status", "accepted", "version", version)).build();
  }
  
  @GET
  @Path("export-data")
  @Authenticated
  @RateLimit(requests = 1, windowSec = 86400, identifier = "user")
  public Response exportData(@Context SecurityContext sec, @Context HttpServletRequest req) {
    Long userId = (Long) sec.getUserPrincipal();
    String exportId = UUID.randomUUID().toString();
    
    Map<String, Object> data = legalService.exportUserData(userId);
    legalService.logExport(userId, exportId, req);
    
    return Response.ok(data).build();
  }
}

@Path("api/v1/legal")
@Produces(MediaType.APPLICATION_JSON)
public class DpoResource {
  
  @Inject DpoContactRepository dpoRepo;
  
  @POST
  @Path("contact-dpo")
  @RateLimit(requests = 1, windowSec = 30, identifier = "ip")
  public Response contactDpo(DpoContactRequest request, @Context HttpServletRequest req) {
    DpoContactRequest dpo = new DpoContactRequest();
    dpo.setName(request.getName());
    dpo.setEmail(request.getEmail());
    dpo.setSubject(request.getSubject());
    dpo.setMessage(request.getMessage());
    dpo.setIpAddress(extractClientIp(req));
    dpo.setStatus("NEW");
    dpoRepo.persist(dpo);
    
    return Response.ok(Map.of(
      "status", "received",
      "message", "DPO responderá en 5 días hábiles"
    )).build();
  }
}
```

**Acceptance Criteria:**

### T-LEGAL-01.8: Modify AuthGuard to check terms acceptance

**Description:** Before redirecting to dashboard, check if terms need acceptance.

**File:** `frontend/app/src/app/auth/auth.guard.ts`

**Code:**
```typescript
@Injectable()
export class AuthGuard implements CanActivate {
  
  constructor(private authService: AuthService, private router: Router) {}
  
  canActivate(route: ActivatedRouteSnapshot): Observable<boolean> {
    return this.authService.getCurrentUser().pipe(
      switchMap(user => 
        this.checkTermsAcceptance(user).pipe(
          map(needsAcceptance => {
            if (needsAcceptance) {
              this.router.navigate(['/terms-acceptance-modal']);
              return false;
            }
            return true;
          })
        )
      )
    );
  }
  
  private checkTermsAcceptance(user: User): Observable<boolean> {
    return this.legalService.getCurrentTermsVersion().pipe(
      map(version => user.accepted_terms_version !== version)
    );
  }
}
```

**Acceptance Criteria:**
- [ ] AuthGuard calls GET /api/v1/legal/current-version
- [ ] Compares user.accepted_terms_version with current version
- [ ] If mismatch: navigate to /terms-acceptance-modal
- [ ] If match: allow navigation to protected route
- [ ] Guard prevents dashboard access until terms accepted
